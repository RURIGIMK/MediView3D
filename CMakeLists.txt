cmake_minimum_required(VERSION 3.16)
project(MediView3D LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(mediview_algorithms src/graphics_algorithms.cpp)
target_include_directories(mediview_algorithms PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_executable(mediview_algorithm_tests tests/algorithm_tests.cpp)
target_link_libraries(mediview_algorithm_tests PRIVATE mediview_algorithms)
enable_testing()
add_test(NAME graphics_algorithms COMMAND mediview_algorithm_tests)

option(MEDIVIEW_BUILD_OPENGL_DEMO "Build the interactive OpenGL checkpoint demo" ON)
if(MEDIVIEW_BUILD_OPENGL_DEMO)
    find_package(OpenGL QUIET)
    find_package(glfw3 QUIET)
    if(OpenGL_FOUND AND TARGET glfw)
        add_executable(mediview_demo src/main.cpp)
        target_link_libraries(mediview_demo PRIVATE mediview_algorithms glfw OpenGL::GL)
    elseif(OpenGL_FOUND AND TARGET glfw3::glfw)
        add_executable(mediview_demo src/main.cpp)
        target_link_libraries(mediview_demo PRIVATE mediview_algorithms glfw3::glfw OpenGL::GL)
    else()
        message(WARNING "OpenGL/GLFW not found. Building the pure C++ algorithms and tests only.")
    endif()
endif()
